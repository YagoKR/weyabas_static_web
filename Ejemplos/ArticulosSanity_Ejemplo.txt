export interface Article {
    title: string;
    slug: string;
    excerpt: string;
    category: string;
    showTag: string;
    imageUrl: string;
    date: string;
    readTime: string;
}

export function formatArticle(raw: any): Article {
    const wordCount = (raw.content ?? [])
        .flatMap((b: any) => b.children?.map((c: any) => c.text) ?? [])
        .join(' ').split(/\s+/).length;

    return {
        title: raw.title,
        slug: raw.slug.current,
        excerpt: raw.excerpt,
        category: raw.category,
        showTag: raw.showTag ?? raw.category,
        imageUrl: raw.image?.asset?.url ?? '',
        date: new Date(raw.publishedAt).toLocaleDateString('en-US', { month: 'long', day: 'numeric' }),
        readTime: `${Math.ceil(wordCount / 200)} min`,
    };
}

const tagClassMap: Record<string, string> = {
    series: 'tag-series',
    anime: 'tag-anime',
    movies: 'tag-movies',
    tech: 'tag-tech',
    deals: 'tag-deals',
};

export function getTagClass(category: string) {
    return tagClassMap[category] ?? 'tag-series';
}
