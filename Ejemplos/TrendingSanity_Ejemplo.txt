// schemas/article.ts (añadir campo)
{ name: 'views', title: 'Views', type: 'number', initialValue: 0 }

---
// src/pages/articles/[slug].astro
import { client } from '../../lib/sanity';

const { slug } = Astro.params;

const article = await client.fetch(
  `*[_type == "article" && slug.current == $slug][0]`,
  { slug }
);

// Incrementar views (fire-and-forget, no bloquea el render)
client
  .patch(article._id)
  .inc({ views: 1 })
  .commit()
  .catch(e => console.error('Error updating views:', e));
---

---
// Trending.astro
import { client } from '../lib/sanity';

const trending = await client.fetch(`
  *[_type == "article"] | order(views desc) [0..4] {
    title, "slug": slug.current
  }
`);
---

<div class="sidebar-widget">
  <div class="widget-title">Trending Now</div>
  {trending.map((item, i) => (
    <a href={`/articles/${item.slug}`} class="trending-item">
      <div class="trending-num">{String(i + 1).padStart(2, '0')}</div>
      <div class="trending-text">{item.title}</div>
    </a>
  ))}
</div>
